\subsection{随机变量及分布}

\subsubsection{离散型随机变量}

\item 已知离散型随机变量 $X$ 服从参数为 $3$ 的泊松分布,则 $P\{X=0\}=(\quad)$
\begin{choices}
\item $e^{-3}$
\item $e^{3}$
\item $e^{-2}$
\item $e^{2}$
\end{choices}

\begin{center}
\begin{tabular}{c|cccc}
\toprule
$X$ & $0$ & $1$ & $2$ & $3$ \\
\midrule
$P$ & $0.2$ & $0.3$ & $k$ & $0.1$ \\
\bottomrule
\end{tabular}
\end{center}
则 $k=(\quad)$
\begin{choices}
\item $0.1$
\item $0.2$
\item $0.3$
\item $0.4$
\end{choices}
\item 设离散型随机变量 $X$ 的分布律为
\begin{center}
\begin{tabular}{c|ccc}
\toprule
$X$ & $-1$ & $0$ & $1$ \\
\midrule
$P$ & $2C$ & $0.4$ & $C$ \\
\bottomrule
\end{tabular}
\end{center}
则常数 $C=(\quad)$
\begin{choices}
\item $0.1$
\item $0.2$
\item $0.3$
\item $0.4$
\end{choices}
\item 已知随机变量 $X$ 服从参数为 $\lambda$ 的泊松分布,且 $P\{X=0\}=e^{-1}$,则 $\lambda=(\quad)$
\begin{choices}

\item $2$
\item $3$
\item $4$
\end{choices}
\item 设随机变量 $X$ 表示 $4$ 次独立重复射击命中目标的次数,每次命中目标的概率为 $0.5$,则 $X$ 服从 $(\quad)$
\begin{choices}
\item $(0\text{-}1)$ 分布
\item $N(4,0.5)$
\item $B(4,0.5)$
\item $U(4,0.5)$
\end{choices}
\item 已知随机变量 $X$ 只能取 $-1$、$0$、$1$、$2$,其相应概率依次为 $\dfrac{1}{2c}$、$\dfrac{3}{4c}$、$\dfrac{5}{8c}$、$\dfrac{7}{16c}$,则 $P\{X<1 \mid X\neq 0\}=(\quad)$
\begin{choices}

\item $\dfrac{8}{25}$
\item $\dfrac{12}{25}$
\item $\dfrac{16}{25}$
\end{choices}
\item 已知离散型随机变量 $X$ 的概率分布如下表所示：

\bottomrule
\end{tabular}
\end{center}
则下列概率计算结果正确的是 $(\quad)$
\begin{choices}

\item $P\{X=0\}=0$
\item $P\{X>-1\}=1$
\item $P\{X<4\}=1$
\end{choices}
\item 设随机变量 $X\sim B(2,p)$,$Y\sim B(3,p)$,若 $P\{X\ge 1\}=\dfrac{5}{9}$,则 $P\{Y\ge 1\}=(\quad)$
\begin{choices}
\item $\dfrac{19}{27}$
\item $\dfrac{5}{27}$
\item $\dfrac{2}{3}$
\item $\dfrac{1}{3}$
\end{choices}
\item 已知随机变量 $X$ 的分布律为
\begin{center}
\begin{tabular}{c|ccc}
\toprule
$X$ & $-1$ & $2$ & $5$ \\

$P$ & $0.2$ & $0.35$ & $0.45$ \\

\begin{choices}

\subsubsection{连续型随机变量}

\begin{enumerate}

\item 已知随机变量 $X$ 的分布函数为
\[
F(x)=
\begin{cases}
0, & x<0,\\[2pt]
\dfrac{1}{2}, & 0\le x<1,\\[6pt]
\dfrac{2}{3}, & 1\le x<3,\\[6pt]
1, & x\ge 3,
\end{cases}
\]
则 $P\{2<X\le 4\}=(\quad)$.
\begin{choices}
  \item $\dfrac{1}{3}$
  \item $\dfrac{2}{3}$
  \item $0$
  \item $1$
\end{choices}

\item 设随机变量 $X$ 概率密度为
\[
f(x)=
\begin{cases}
24x^2, & 0\le x\le a,\\
0, & \text{其他},
\end{cases}
\]
则常数 $a=(\quad)$.
\begin{choices}
  \item $1$
  \item $\dfrac{1}{2}$
  \item $\dfrac{1}{3}$
  \item $\dfrac{1}{4}$
\end{choices}

\item 设连续型随机变量的分布函数为
\[
F(x)=
\begin{cases}
0, & x<0,\\[4pt]
\sin x, & 0\le x<\dfrac{\pi}{2},\\[6pt]
1, & x\ge \dfrac{\pi}{2},
\end{cases}
\]
其概率密度为 $f(x)$,则 $f\!\left(\dfrac{\pi}{6}\right)=(\quad)$.
\begin{choices}
  \item $\dfrac{1}{3}$
  \item $\dfrac{\sqrt{3}}{2}$
  \item $\dfrac{\sqrt{3}}{3}$
  \item $\dfrac{1}{2}$
\end{choices}

\item 设随机变量 $X$ 的概率分布为：
\begin{center}
\begin{tabular}{cccccc}
\toprule
$X$ & $1$ & $2$ & $3$ & $4$ \\
\midrule
$P$ & $\dfrac{1}{4}$ & $\dfrac{1}{8}$ & $\dfrac{4}{7}$ & $\dfrac{3}{56}$ \\
\bottomrule
\end{tabular}
\end{center}
$F(x)$ 为其分布函数,则 $F(3)=(\quad)$.
\begin{choices}
  \item $\dfrac{1}{4}$
  \item $\dfrac{4}{7}$
  \item $\dfrac{53}{56}$
  \item $\dfrac{3}{56}$
\end{choices}

\item 已知 $F_1(x)$,$F_2(x)$ 都是一维随机变量的分布函数,则下列函数中仍可作为一维随机变量分布函数的是 $(\quad)$.
\begin{choices}
  \item $F_1(x)-F_2(x)$
  \item $\dfrac{1}{2}F_1(x)+\dfrac{1}{2}F_2(x)$
  \item $\dfrac{1}{2}F_1(x)-\dfrac{1}{2}F_2(x)$
  \item $F_1(x)+F_2(x)$
\end{choices}

\item 设随机变量 $X$ 的概率密度为
\[
f(x)=
\begin{cases}
ke^{-\frac{x}{3}}, & x\ge 0,\\
0, & x<0,
\end{cases}
\]
则常数 $k=(\quad)$.
\begin{choices}
  \item $-\dfrac{1}{5}$
  \item $\dfrac{1}{5}$
  \item $1$
  \item $5$
\end{choices}

\item 已知随机变量 $X$ 的概率密度为 $f(x)=ce^{-|x|},\,-\infty<x<+\infty$,则 $c=(\quad)$.
\begin{choices}
  \item $\dfrac{1}{3}$
  \item $-\dfrac{1}{3}$
  \item $\dfrac{1}{2}$
  \item $-\dfrac{1}{2}$
\end{choices}

\item 设离散型随机变量 $X$ 的分布函数为
\[
F(x)=
\begin{cases}
0, & x<-1,\\[4pt]
\dfrac{1}{3}, & -1\le x<2,\\[6pt]
1, & x\ge 2,
\end{cases}
\]
则 $P\{X=2\}=(\quad)$.
\begin{choices}
  \item $\dfrac{1}{3}$
  \item $\dfrac{2}{3}$
  \item $0$
  \item $1$
\end{choices}

\item 下列函数中可作为随机变量分布函数的是 $(\quad)$.
\begin{choices}
  \item $F(x)=
    \begin{cases}
    1, & 0\le x\le 1,\\
    0, & \text{其他}
    \end{cases}$
  \item $F(x)=
    \begin{cases}
    -1, & x<0,\\
    x, & 0\le x<1,\\
    0, & x\ge 1
    \end{cases}$
  \item $F(x)=
    \begin{cases}
    0, & x<0,\\
    x, & 0\le x<1,\\
    1, & x\ge 1
    \end{cases}$
  \item $F(x)=
    \begin{cases}
    0, & x<0,\\
    x, & 0\le x<1,\\
    2, & x\ge 1
    \end{cases}$
\end{choices}

\item 设随机变量 $X$ 的概率密度为
\[
f(x)=
\begin{cases}
x, & 0<x\le 1,\\
2-x, & 1<x\le 2,\\
0, & \text{其他},
\end{cases}
\]
则 $P\{0.2<X<1.2\}=(\quad)$.
\begin{choices}
  \item $0.5$
  \item $0.6$
  \item $0.66$
  \item $0.7$
\end{choices}

\item 设函数 $f(x)$ 在 $[a,b]$ 上等于 $\sin x$,在此区间外等于零.若 $f(x)$ 可以作为某连续型随机变量的概率密度,则区间 $[a,b]$ 应为 $(\quad)$.
\begin{choices}
  \item $\left[-\dfrac{\pi}{2},0\right]$
  \item $\left[0,\dfrac{\pi}{2}\right]$
  \item $\left[0,\pi\right]$
  \item $\left[0,\dfrac{3\pi}{2}\right]$
\end{choices}

\end{enumerate}


\subsubsection{随机变量函数的分布}

\begin{enumerate}
  \item 设随机变量 $X$ 的概率密度为 $f(x)=\dfrac{1}{2}e^{-|x|},\,-\infty<x<+\infty$,则 $P\{0<X<1\}=(\quad)$.
  \begin{choices}
    \item $\dfrac{1}{2}$
    \item $\dfrac{1}{3}$
    \item $\dfrac{1}{2}+\dfrac{1}{2e}$
    \item $\dfrac{1}{2}-\dfrac{1}{2e}$
  \end{choices}

  \item 设随机变量 $X\sim N(0,1)$,概率密度为 $f(x)$,分布函数为 $F(x)$,则下列错误的是
  \begin{choices}
    \item $P\{X\le 0\}=P\{X\ge 0\}$
    \item $P\{X\le 1\}=P\{X\ge 1\}$
    \item $f(x)=f(-x),\ x\in R$
    \item $F(x)=1-F(-x),\ x\in R$
  \end{choices}

  \item 已知连续型随机变量 $X$ 服从区间 $[a,b]$ 上的均匀分布,则概率 $P\left\{X<\dfrac{2a+b}{3}\right\}=(\quad)$.
  \begin{choices}
    \item $0$
    \item $\dfrac{1}{3}$
    \item $\dfrac{2}{3}$
    \item $1$
  \end{choices}

  \item 设随机变量 $X$ 的密度函数为 $f(x)=e^{-x},\,x>0$,则 $Y=2X+1$ 的密度函数为
  \begin{choices}
    \item $e^{-\frac{y-1}{2}},\ y>1$
    \item $\dfrac{1}{2}e^{-\frac{y-1}{2}},\ y>1$
    \item $\dfrac{1}{2}e^{-\frac{y+1}{2}},\ y>1$
    \item $e^{-\frac{y+1}{2}},\ y>1$
  \end{choices}

  \item 设随机变量 $X$ 的分布函数为 $F(x)$,则随机变量 $Y=2X-2$ 的分布函数为 $(\quad)$.
  \begin{choices}
    \item $F\left(\dfrac{1}{2}y-\dfrac{1}{2}\right)$
    \item $F\left(\dfrac{1}{2}y+1\right)$
    \item $2F(y)+1$
    \item $\dfrac{1}{2}F(y)-\dfrac{1}{2}$
  \end{choices}

  \item 设随机变量 $X\sim N(0,4)$,则 $P\{X\ge 0\}=(\quad)$.
  \begin{choices}
    \item $1$
    \item $\dfrac{1}{2}$
    \item $\dfrac{1}{3}$
    \item $\dfrac{1}{4}$
  \end{choices}

  \item 设随机变量 $X\sim N(2,\sigma^2)$,且 $P\{2\le X\le 4\}=0.3$,则 $P\{X\le 0\}=(\quad)$.
  \begin{choices}
    \item $0.1$
    \item $0.2$
    \item $0.3$
    \item $0.4$
  \end{choices}

  \item 设随机变量 $X$ 服从参数为 $3$ 的指数分布,其分布函数为 $F(x)$,则 $F(3)=(\quad)$.
  \begin{choices}
    \item $\dfrac{1}{3e}$
    \item $\dfrac{e}{3}$
    \item $1-e^{-1}$
    \item $1-\dfrac{1}{3}e^{-1}$
  \end{choices}

  \item 设随机变量 $X\sim U(0,5)$,随机变量 $X$ 的分布函数为 $(\quad)$.
  \begin{choices}
    \item $F(x)=\begin{cases} 0, & x<0 \\[2pt] \dfrac{1}{5}, & 0\le x<5 \\[2pt] 1, & x\ge 5 \end{cases}$
    \item $F(x)=\begin{cases} 0, & x<0 \\[2pt] \dfrac{x}{5}, & 0\le x<5 \\[2pt] 1, & x\ge 5 \end{cases}$
    \item $F(x)=\begin{cases} 0, & x<0 \\[2pt] \dfrac{1-x}{5}, & 0\le x<5 \\[2pt] 1, & x\ge 5 \end{cases}$
    \item $F(x)=\begin{cases} 0, & x<0 \\[2pt] \dfrac{x-1}{5}, & 0\le x<5 \\[2pt] 1, & x\ge 5 \end{cases}$
  \end{choices}

  \item 设随机变量 $X\sim N(1,4)$,$\Phi(x)$ 为标准正态分布函数,已知 $\Phi(1)=0.8413$,$\Phi(0)=0.5$,$\Phi(3)=0.9987$,$\Phi(0.5)=0.6915$,则事件 $\{1\le X\le 3\}$ 的概率为 $(\quad)$.
  \begin{choices}
    \item $0.1574$
    \item $0.1915$
    \item $0.4987$
    \item $0.3413$
  \end{choices}

  \item 设随机变量 $X$ 服从区间 $[-2,2]$ 上的均匀分布,则 $Y=X^{2}$ 的概率密度函数为
  \begin{choices}
    \item $f_Y(y)=\begin{cases} \dfrac{1}{4\sqrt{y}}, & 0<y\le 2 \\[2pt] 0, & \text{其它} \end{cases}$
    \item $f_Y(y)=\begin{cases} \dfrac{1}{4\sqrt{y}}, & 0<y\le 4 \\[2pt] 0, & \text{其它} \end{cases}$
    \item $f_Y(y)=\begin{cases} \dfrac{1}{2\sqrt{y}}, & 0<y\le 4 \\[2pt] 0, & \text{其它} \end{cases}$
    \item $f_Y(y)=\begin{cases} \dfrac{1}{4}, & 0<y\le 4 \\[2pt] 0, & \text{其它} \end{cases}$
  \end{choices}

  \item 设随机变量 $X$ 的概率密度函数为 $f_X(x)$,则 $Y=-2X+3$ 的密度函数为 $(\quad)$.
  \begin{choices}
    \item $\dfrac{1}{2}f_X\left(-\dfrac{y-3}{2}\right)$
    \item $\dfrac{1}{2}f_X\left(-\dfrac{y+3}{2}\right)$
    \item $-\dfrac{1}{2}f_X\left(-\dfrac{y-3}{2}\right)$
    \item $-\dfrac{1}{2}f_X\left(-\dfrac{y+3}{2}\right)$
  \end{choices}

  \item 设随机变量 $X$ 的分布律如表所示,$Y=X^{2}$,则 $P\{Y=1\}=(\quad)$.
  \begin{center}
    \begin{tabular}{c c c c}
      \toprule
      $X$ & $-1$ & $0$ & $1$ \\
      \midrule
      $P$ & $\dfrac{1}{2}$ & $\dfrac{1}{3}$ & $\dfrac{1}{6}$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{choices}
    \item $\dfrac{1}{2}$
    \item $\dfrac{1}{6}$
    \item $\dfrac{2}{3}$
    \item $\dfrac{1}{3}$
  \end{choices}

  \item 设随机变量 $X$ 的分布律为
  \begin{center}
    \begin{tabular}{c c c c c}
      \toprule
      $X$ & $-1$ & $0$ & $1$ & $2$ \\
      \midrule
      $P$ & $\dfrac{1}{8}$ & $\dfrac{3}{8}$ & $\dfrac{1}{16}$ & $\dfrac{7}{16}$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  且 $Y=X^{2}$,记随机变量 $Y$ 的分布函数为 $F_Y(y)$,则 $F_Y(3)=(\quad)$.
  \begin{choices}
    \item $\dfrac{1}{16}$
    \item $\dfrac{7}{16}$
    \item $\dfrac{9}{16}$
    \item $\dfrac{3}{16}$
  \end{choices}

  \item 设连续型随机变量 $X\sim N(1,4)$,则 $\dfrac{X-1}{2}\sim(\quad)$.
  \begin{choices}
    \item $N(0,2)$
    \item $N(1,2)$
    \item $N(2,4)$
    \item $N(0,1)$
  \end{choices}
\end{enumerate}
