\subsection{概率论的基本概念}

\subsubsection{随机事件}

\begin{enumerate}
    \item  设事件 $A$ 与 $B$ 同时发生时 $C$ 也发生,则(\quad).
    \begin{choices}
        \item $A\cup B$ 是 $C$ 的子事件；
        \item $\overline{ABC}$；或 $\overline{A}\cup\overline{B}\cup\overline{C}$；
        \item $AB$ 是 $C$ 的子事件；
        \item $C$ 是 $AB$ 的子事件.
    \end{choices}

    \item  若 $AB$ 为不可能事件,则事件 $A$ 和 $B$(\quad).
    \begin{choices}
        \item 同时发生
        \item 互不相容
        \item 互逆
        \item 相等
    \end{choices}

    \item  设事件 $A=\{\text{甲种产品畅销或乙种产品滞销}\}$,则 $A$ 的对立事件为(\quad).
    \begin{choices}
        \item 甲种产品滞销,乙种产品畅销；
        \item 甲种产品滞销；
        \item 甲、乙两种产品均畅销；
        \item 甲种产品滞销或者乙种产品畅销.
    \end{choices}

    \item  德摩根律 $\overline{A\cap B}=$(\quad).
    \begin{choices}
        \item $\overline{A}\cup\overline{B}$
        \item $\overline{A\cap B}$
        \item $A\cap B$
        \item $\overline{A}\cap\overline{B}$
    \end{choices}

    \item  若对事件 $A,B$ 有 $B\subset A$,则下列命题正确的是(\quad).
    \begin{choices}
        \item $A,B$ 必同时成立
        \item $A$ 发生,$B$ 必发生
        \item $A$ 不发生,$B$ 必不发生
        \item $B$ 不发生,$A$ 必不发生
    \end{choices}

    \item  下面等式不成立的是(\quad).
    \begin{choices}
        \item $AB\cup A\overline{B}=A$
        \item $\overline{AB}=A\cup B$
        \item $(AB)(\overline{A}\,\overline{B})=\varnothing$
        \item $B\cup\overline{AB}=\overline{A}\cup B$
    \end{choices}

    \item  如果(\quad)成立,则事件 $A$ 与 $B$ 互为逆事件.
    \begin{choices}
        \item $AB=\varnothing$
        \item $A\cup B=S$
        \item $AB=\varnothing$ 且 $A\cup B=S$
        \item $A$ 与 $B$ 互不相容
    \end{choices}

    \item  甲、乙两人进行象棋比赛,设 $A$ 表示“甲胜乙负”,则 $\overline{A}$ 表示(\quad).
    \begin{choices}
        \item 甲胜乙负
        \item 甲乙平局
        \item 甲负
        \item 甲负或平局
    \end{choices}

    \item  掷一颗骰子,观察出现的点数.$A$ 表示“出现奇数点”,$B$ 表示“出现的点数大于等于 $5$”,则 $A\cap B=$(\quad).
    \begin{choices}
        \item $\{1,3,5\}$
        \item $\{5,6\}$
        \item $\{5\}$
        \item $\{1\}$ 或 $\{3\}$
    \end{choices}

    \item  掷一颗骰子,观察出现的点数.$A$ 表示“出现奇数点”,$B$ 表示“出现的点数大于等于 $5$”,则 $A-B=$(\quad).
    \begin{choices}
        \item $\{1,3,5\}$
        \item $\{5,6\}$
        \item $\{5\}$
        \item $\{1,3\}$
    \end{choices}
\end{enumerate}

\subsubsection{概率}

\begin{enumerate}
    \item  已知 $P(\overline{A})=0.5$,$P(\overline{A}B)=0.2$,$P(B)=0.4$,则 $P(AB)=$(\quad)
    \begin{choices}
        \item $0.2$
        \item $0.3$
        \item $0.7$
        \item $0.8$
    \end{choices}

    \item  袋中共有5个球,其中3个红球、2个白球,每次取一个,无放回地取2次,则第二次取到红球的概率是(\quad)
    \begin{choices}
        \item $\dfrac{3}{5}$
        \item $\dfrac{3}{4}$
        \item $\dfrac{2}{4}$
        \item $\dfrac{3}{10}$
    \end{choices}

    \item  设 $P(A)=0.1$,$P(A\cup B)=0.3$,且 $A$ 与 $B$ 互不相容,则 $P(B)=(\quad)$
    \begin{choices}
        \item $0.2$
        \item $0.5$
        \item $0.4$
        \item $0.1$
    \end{choices}

    \item  设 $A,B$ 是任意两个事件,$P(A)=0.5$,$P(A-B)=0.2$,则 $P(\overline{AB})=(\quad)$
    \begin{choices}
        \item $0.3$
        \item $0.85$
        \item $0.7$
        \item $0.1$
    \end{choices}

    \item  设 $A,B,C$ 为三个事件,下面正确的是(\quad)
    \begin{choices}
        \item $P(A \overline{B})=1-P(AB)$
        \item $P(A\cup\overline{B})=P(A)-P(B)+1$
        \item $P(ABC)=1-P(\overline{A}\overline{B}\overline{C})$
        \item $P(A-B)=P(A \overline{B})$
    \end{choices}

    \item  在区间 $(0,1)$ 中随机地取两个数,求两数之差的绝对值小于$\frac{1}{2}$的概率是(\quad)
    \begin{choices}
        \item $\dfrac{3}{5}$
        \item $\dfrac{3}{4}$
        \item $\dfrac{2}{4}$
        \item $\dfrac{3}{10}$
    \end{choices}

    \item  已知 $P(\overline{A})=0.5$,$P(\overline{A}B)=0.2$,$P(B)=0.4$,则 $P(A-B)=(\quad)$
    \begin{choices}
        \item $0.2$
        \item $0.3$
        \item $0.7$
        \item $0.8$
    \end{choices}

    \item  设 $A,B$ 是任意两个事件,且 $P(AB)=0$,则有(\quad)
    \begin{choices}
        \item $A$ 与 $B$ 互不相容
        \item $AB$ 是不可能事件
        \item $AB$ 未必是不可能事件
        \item $A$ 与 $B$ 互为逆事件
    \end{choices}

    \item  设 $A,B$ 是任意两个概率不为零的不相容事件,则下列结论一定正确的是(\quad)
    \begin{choices}
        \item $\overline{A}$ 与 $\overline{B}$ 不相容
        \item $\overline{A}$ 与 $\overline{B}$ 相容
        \item $P(A-B)=P(A)-P(B)$
        \item $P(A-B)=P(A)$
    \end{choices}

    \item  设 $A,B$ 是任意两个事件,则 $P(A-B)=(\quad)$
    \begin{choices}
        \item $P(A)-P(B)$
        \item $P(A)-P(AB)$
        \item $P(A)-P(B)-P(\overline{A}B)$
        \item $P(A)+P(\overline{B})-P(AB)$
    \end{choices}
\end{enumerate}

\subsubsection{条件概率与独立性}

\begin{enumerate}
  \item  假设一批产品中一、二、三等品各占 $50\%$、$20\%$、$30\%$,从中随机取出一件,结果不是三等品,则取到的一等品的概率为(\quad)．
  \begin{choices}
    \item $\dfrac{3}{7}$
    \item $\dfrac{2}{7}$
    \item $\dfrac{5}{7}$
    \item $\dfrac{7}{10}$
  \end{choices}

  \item  设某公路上经过的货车与客车数量之比为 $2:1$,货车中途停车修理的概率为 $0.02$,客车停车修理的概率为 $0.01$,则中途有车辆停车修理的概率为(\quad)
  \begin{choices}
    \item $\dfrac{1}{60}$
    \item $\dfrac{1}{30}$
    \item $\dfrac{1}{5}$
    \item $\dfrac{1}{6}$
  \end{choices}

  \item  设 $A,B$ 为随机事件,且 $P(B)>0$,$P(A\mid B)=1$,则必有(\quad)
  \begin{choices}
    \item $P(A\cup B)>P(A)$
    \item $P(A\cup B)>P(B)$
    \item $P(A\cup B)=P(A)$
    \item $P(A\cup B)=P(B)$
  \end{choices}

  \item  某篮球队员在比赛中每次罚球的命中率相同,且在两次罚球中至多命中一次的概率为 $\dfrac{16}{25}$,则该队员每次罚球的命中率为(\quad)．
  \begin{choices}
    \item $\dfrac{1}{5}$
    \item $\dfrac{2}{5}$
    \item $\dfrac{3}{5}$
    \item $\dfrac{4}{5}$
  \end{choices}

  \item  种植两株不同的花卉,若它们的成活率分别为 $p$ 和 $q$,则恰好有一株成活的概率为(\quad)
  \begin{choices}
    \item $p+q-2pq$
    \item $p+q-pq$
    \item $p+q$
    \item $p$
  \end{choices}

  \item  来成都旅游的外地游客中,若甲、乙、丙三人选择去武侯祠游览的概率均为 $\dfrac{3}{5}$,且他们的选择互不影响,则这三人中至多两人选择去武侯祠游览的概率为(\quad)
  \begin{choices}
    \item $\dfrac{36}{125}$
    \item $\dfrac{44}{125}$
    \item $\dfrac{54}{125}$
    \item $\dfrac{98}{125}$
  \end{choices}

  \item  设 $A,B$ 为任意两个事件,若 $P(A)=0.5$,$P(B\mid A)=0.8$,则 $P(AB)=$(\quad)
  \begin{choices}
    \item $0.1$
    \item $0.2$
    \item $0.3$
    \item $0.4$
  \end{choices}

  \item  设事件 $A,B$ 相互独立,则(\quad)
  \begin{choices}
    \item $P(B\mid A)=0$
    \item $P(AB)=P(A)P(B)$
    \item $P(B\mid A)=P(B)$
    \item $P(A\mid B)=P(A)$
  \end{choices}

  \item  某人提出一个问题,甲先答,答对的概率为 $0.4$,如果甲答错,由乙答,答对的概率为 $0.5$,则问题由乙答对的概率为(\quad)．
  \begin{choices}
    \item $0.6$
    \item $0.5$
    \item $0.3$
    \item $0.4$
  \end{choices}

  \item  在 $10$ 个形状大小均相同的球中有 $6$ 个红球和 $4$ 个白球,不放回地依次摸出 $2$ 个球,在第 $1$ 次摸出红球的条件下,第 $2$ 次也摸到红球的概率为(\quad)．
  \begin{choices}
    \item $\dfrac{6}{10}$
    \item $\dfrac{5}{9}$
    \item $\dfrac{30}{90}$
    \item $\dfrac{36}{90}$
  \end{choices}
\end{enumerate}
